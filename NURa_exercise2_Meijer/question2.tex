photoionization

\Gamma_\mathrm{pe} = \alpha_\mathrm{B} n_\mathrm{H} n_\mathrm{e} \psi k_\mathrm{B} T_\mathrm{c}
\end{equation}


radiative recombination
\begin{equation}\label{eqn:cooling}
\Lambda_\mathrm{rr} = \alpha_\mathrm{B} n_\mathrm{H} n_\mathrm{e} \langle E_\mathrm{rr} \rangle
\end{equation}



\begin{equation}
\begin{split}
\Gamma_\mathrm{pe} - \Lambda_\mathrm{rr} &= \psi k_\mathrm{B} T_\mathrm{c} - \langle E_\mathrm{rr} \rangle \\
&= \psi T_\mathrm{c} - \left[ 0.684 - 0.0416 \ln\left(\frac{T_4}{Z^2}\right) \right] T \\
&= 0
\end{split}
\end{equation}

[find the T where this is true]


Analytical result
\begin{equation}
\begin{split}
\psi k_\mathrm{B} T_\mathrm{c} - \left[ 0.684 - 0.0416 \ln(\frac{T_4}{Z^2}) \right] k_\mathrm{B}T &= 0 
\\
\left[ 0.684 - 0.0416 \ln(\frac{T_4}{Z^2}) \right] T &= \psi T_\mathrm{c} 
\\
\left[ 0.684 - 0.0416 \ln(\frac{T_4}{Z^2}) \right] &= \frac{\psi T_\mathrm{c}}{T}
\\
\ln(\frac{T_4}{Z^2}) &= \left[ 0.684 - \frac{\psi T_\mathrm{c}}{T} \right] \frac1{0.0416}
\\
\end{split}
\end{equation}
